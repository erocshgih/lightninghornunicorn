            <script>
            (function initDonateCoinGame() {
              var stage = document.getElementById('lhu-donate-stage');
              var sticker = document.getElementById('lhu-donate-sticker');
              var form = document.getElementById('lhu-donate-form');
              var closeup = document.getElementById('lhu-donate-closeup');
              var heartEyes = document.getElementById('lhu-donate-hearteyes');
              var stickerImg = document.getElementById('lhu-donate-img');
              var counter = document.getElementById('lhu-donate-counter');
              var status = document.getElementById('lhu-donate-status');
              var fill = document.getElementById('lhu-donate-fill');
              if (!stage || !sticker || !form) return;

              var JACKPOT = 20;
              var clicks = 0, busy = false, finished = false;
              var timers = [], actx = null;
              var reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
              var paypalUrl = form.getAttribute('action');
              // liftdown.png is not delivered yet - fall back to liftup until it is.
              if (stickerImg) stickerImg.addEventListener('error', function () {
                if (this.src.indexOf('liftdown') >= 0) this.src = 'donate/liftup.png';
              });

              function later(fn, ms) { var id = setTimeout(fn, ms); timers.push(id); return id; }
              function ensureAudio() {
                if (!actx) {
                  var AC = window.AudioContext || window.webkitAudioContext;
                  if (!AC) return null;
                  try { actx = new AC(); } catch (e) { return null; }
                }
                if (actx.state === 'suspended') actx.resume();
                return actx;
              }
              function playBoing() {
                var ctx = ensureAudio(); if (!ctx) return;
                var t = ctx.currentTime;
                var osc = ctx.createOscillator(), g = ctx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(680, t);
                osc.frequency.exponentialRampToValueAtTime(130, t + 0.26);
                g.gain.setValueAtTime(0.0001, t);
                g.gain.exponentialRampToValueAtTime(0.24, t + 0.02);
                g.gain.exponentialRampToValueAtTime(0.0001, t + 0.3);
                osc.connect(g); g.connect(ctx.destination);
                osc.start(t); osc.stop(t + 0.32);
              }
              function playApplause(done) {
                var ctx = ensureAudio(), dur = 2.0;
                if (!ctx) { later(done, 2100); return; }
                var t0 = ctx.currentTime, len = Math.ceil(ctx.sampleRate * dur);
                var buf = ctx.createBuffer(1, len, ctx.sampleRate), d = buf.getChannelData(0);
                for (var i = 0; i < len; i++) d[i] = Math.random() * 2 - 1;
                var src = ctx.createBufferSource(); src.buffer = buf;
                var filt = ctx.createBiquadFilter(); filt.type = 'bandpass'; filt.frequency.value = 1700; filt.Q.value = 0.7;
                var gain = ctx.createGain(); gain.gain.value = 0.0001;
                for (var k = 0; k < 20; k++) {
                  var tt = t0 + k * 0.095;
                  gain.gain.setValueAtTime(0.0001, tt);
                  gain.gain.linearRampToValueAtTime(0.16 + Math.random() * 0.09, tt + 0.006);
                  gain.gain.exponentialRampToValueAtTime(0.0001, tt + 0.075);
                }
                src.connect(filt); filt.connect(gain); gain.connect(ctx.destination);
                // Fire only when the audio buffer really ends - not on a guessed timer.
                src.onended = done;
                src.start(t0); src.stop(t0 + dur);
              }
              function throwCoin() {
                if (reduce) return;
                var coin = document.createElement('img');
                coin.className = 'lhu-donate-coin';
                coin.src = 'donate/coin.png';
                coin.alt = ''; coin.setAttribute('aria-hidden', 'true');
                var w = stage.clientWidth, h = stage.clientHeight;
                var sx = w * (0.2 + Math.random() * 0.6);
                var sy = h - 6;
                coin.style.left = (sx - 56) + 'px';
                coin.style.top = (sy - 56) + 'px';
                stage.appendChild(coin);
                var dx = w / 2 - sx, dy = h / 2 - sy;
                var frames = [
                  { transform: 'translate(0,0) scale(0.55) rotate(0deg)', opacity: 0 },
                  { transform: 'translate(' + (dx * 0.55) + 'px,' + (dy * 0.55 - 120) + 'px) scale(1.2) rotate(220deg)', opacity: 1, offset: 0.5 },
                  { transform: 'translate(' + dx + 'px,' + dy + 'px) scale(0.7) rotate(400deg)', opacity: 0 }
                ];
                if (coin.animate) {
                  var anim = coin.animate(frames, { duration: 1150, easing: 'cubic-bezier(.22,.68,.3,1)' });
                  anim.finished.then(function () { coin.remove(); }).catch(function () { coin.remove(); });
                } else {
                  later(function () { coin.remove(); }, 1180);
                }
              }
              var isGerman = (window.LHU_LANGUAGE || 'de') === 'de';
              var stages = isGerman
                ? ['WIRF EINE MÜNZE', 'MEHR!', 'SCHNELLER!', 'DU BIST SO SCHNELL', '¡ARRIBA, ARRIBA!']
                : ['THROW A COIN', 'MORE!', 'FASTER!', 'YOU ARE SO FAST', '¡ARRIBA, ARRIBA!'];
              function stageLabel(n) {
                if (n === 0) return stages[0];
                if (n <= 5) return stages[1];
                if (n <= 10) return stages[2];
                if (n <= 15) return stages[3];
                return stages[4];
              }
              function showCloseup() {
                playBoing();
                closeup.classList.add('show');
                later(function () { closeup.classList.remove('show'); }, 720);
              }
              function swapStickerArt() {
                if (!stickerImg) return;
                var src = clicks >= JACKPOT ? 'donate/winner.png'
                  : (clicks % 2 === 1 ? 'donate/liftup.png' : 'donate/liftdown.png');
                if (stickerImg.getAttribute('src') !== src) stickerImg.src = src;
              }
              function finish() {
                finished = true;
                heartEyes.classList.add('show');
                // Give the visitor the full two seconds on the winning image
                // before the PayPal button is enabled and the link is opened.
                var win = window.open('', '_blank');
                playApplause(function () {
                  sticker.classList.add('is-paypal-ready');
                  sticker.type = 'submit';
                  if (status) status.textContent = isGerman ? 'Jackpot! PayPal ist jetzt verfügbar.' : 'Jackpot! PayPal is ready.';
                  if (win) {
                    try { win.location.href = paypalUrl; }
                    catch (e) { window.open(paypalUrl, '_blank', 'noopener'); }
                  } else {
                    window.open(paypalUrl, '_blank', 'noopener');
                  }
                });
              }
              sticker.addEventListener('click', function (ev) {
                if (finished) return; // native submit takes over after jackpot
                ev.preventDefault();
                if (busy) return;
                clicks++;
                swapStickerArt();
                if (fill) fill.style.transform = 'scaleY(' + (clicks / JACKPOT).toFixed(4) + ')';
                if (counter) counter.innerHTML = stageLabel(clicks) + ' <b>' + clicks + '</b> / ' + JACKPOT;
                if (status) status.textContent = (isGerman ? 'Münze ' : 'Coin ') + clicks + ' / ' + JACKPOT;
                throwCoin();
                sticker.animate && sticker.animate([
                  { transform: 'scale(1)' }, { transform: 'scale(0.93)' }, { transform: 'scale(1)' }
                ], { duration: 180, easing: 'ease-out' });
                if (clicks % 3 === 0) showCloseup();
                if (clicks >= JACKPOT) { busy = true; later(finish, reduce ? 0 : 650); }
              });
              window.__lhuRegisterCleanup && window.__lhuRegisterCleanup(function () {
                timers.forEach(clearTimeout);
                if (actx && actx.state !== 'closed') { try { actx.close(); } catch (e) {} }
                actx = null;
              });
            })();
            </script>

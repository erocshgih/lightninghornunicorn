        <script>

            (function initStorytime() {
                let totalChapters = 0;
                let currentChapter = 1;
                let currentLang = (window.LHU_LANGUAGE || navigator.language || 'en').toLowerCase().startsWith('de') ? 'de' : 'en';
                let chapterData = { en: '', de: '' };

                const contentDiv = document.getElementById('text-content');
                const navDiv = document.getElementById('dynamic-chapter-nav');
                const btnEn = document.getElementById('lang-en');
                const btnDe = document.getElementById('lang-de');
                if (!contentDiv || !navDiv || !btnEn || !btnDe) return;

                if (currentLang === 'de') {
                    btnDe.classList.add('active');
                    btnDe.setAttribute('aria-pressed', 'true');
                    btnEn.classList.remove('active');
                    btnEn.setAttribute('aria-pressed', 'false');
                }

                let disposed = false;
                let startupTimer = null;
                let scanController = null;
                let fetchController = null;
                let chapterLoadToken = 0;
                let loadedChapter = 0;

                function cancelChapterFetch() {
                    if (fetchController) {
                        fetchController.abort();
                        fetchController = null;
                    }
                }

                function cleanup() {
                    disposed = true;
                    chapterLoadToken++;
                    if (startupTimer !== null) {
                        clearTimeout(startupTimer);
                        startupTimer = null;
                    }
                    if (scanController) {
                        scanController.abort();
                        scanController = null;
                    }
                    cancelChapterFetch();
                }

                if (typeof window.__lhuRegisterCleanup === 'function') {
                    window.__lhuRegisterCleanup(cleanup);
                } else {
                    window.addEventListener('pagehide', cleanup, { once: true });
                }

                btnEn.addEventListener('click', () => {
                    if (disposed) return;
                    currentLang = 'en';
                    btnEn.classList.add('active');
                    btnEn.setAttribute('aria-pressed', 'true');
                    btnDe.classList.remove('active');
                    btnDe.setAttribute('aria-pressed', 'false');
                    renderText();
                });

                btnDe.addEventListener('click', () => {
                    if (disposed) return;
                    currentLang = 'de';
                    btnDe.classList.add('active');
                    btnDe.setAttribute('aria-pressed', 'true');
                    btnEn.classList.remove('active');
                    btnEn.setAttribute('aria-pressed', 'false');
                    renderText();
                });

                async function scanChapters() {
                    if (disposed) return;
                    const MAX_CHAPTERS = 200;
                    const cacheKey = 'lhu-storytime-chaptercount';
                    let cachedCount = null;
                    try { cachedCount = sessionStorage.getItem(cacheKey); } catch (_) {   }

                    if (cachedCount !== null) {
                        const parsedCount = Number.parseInt(cachedCount, 10);
                        
                        
                        if (Number.isInteger(parsedCount) && parsedCount >= 0 && parsedCount <= MAX_CHAPTERS) {
                            totalChapters = parsedCount;
                            if (totalChapters > 0) {
                                buildNavigation();
                                loadChapter(1);
                            } else {
                                showEmptyArchive();
                            }
                            return;
                        }
                    }

                    let count = 1;
                    
                    
                    while (!disposed && count <= MAX_CHAPTERS) {
                        const numStr = String(count).padStart(6, '0');
                        const controller = new AbortController();
                        scanController = controller;
                        const scanTimeout = window.setTimeout(() => controller.abort(new DOMException('Chapter scan timed out', 'TimeoutError')), 8000);
                        let response;
                        try {
                            response = await fetch(`chapters/${numStr}.txt`, {
                                method: 'HEAD',
                                signal: controller.signal,
                                cache: 'no-store'
                            });

                            
                            if (response.status === 405 || response.status === 501) {
                                response = await fetch(`chapters/${numStr}.txt`, {
                                    method: 'GET',
                                    signal: controller.signal,
                                    cache: 'no-store'
                                });
                                if (response.body) {
                                    try { await response.body.cancel(); } catch (_) {   }
                                }
                            }

                            if (disposed) return;
                            if (response.status === 404) break;
                            if (!response.ok) throw new Error(`HTTP ${response.status}`);
                            const contentType = (response.headers.get('content-type') || '').toLowerCase();
                            
                            if (contentType.includes('text/html')) break;
                            count++;
                        } catch (error) {
                            if (error.name === 'AbortError' || disposed) return;
                            contentDiv.innerHTML = '<p>CONNECTION ERROR: ARCHIVE COULD NOT BE REACHED.</p>';
                            return;
                        } finally {
                            clearTimeout(scanTimeout);
                            if (scanController === controller) scanController = null;
                        }
                    }

                    if (disposed) return;
                    totalChapters = count - 1;
                    try { sessionStorage.setItem(cacheKey, String(totalChapters)); } catch (_) {   }

                    if (totalChapters > 0) {
                        buildNavigation();
                        loadChapter(1);
                    } else {
                        showEmptyArchive();
                    }
                }

                function showEmptyArchive() {
                    if (disposed) return;
                    contentDiv.innerHTML = `
                    <div style="text-align: center; margin-bottom: 25px;">
                        <span style="color: var(--strip-active); font-size: 24px;">[ ARCHIVE DIRECTORY EMPTY ]</span>
                    </div>
                    <div style="padding: 20px; border: 1px dashed var(--border-soft); border-radius: 8px;">
                        <strong>BACK COVER TEASER // SYSTEM LOG</strong><br><br>
                        [SCENE START]<br><br>
                        INT. NEON-LIT BUNKER IN VIENNA - NIGHT<br><br>
                        VEDRAN (LIGHTNING HORN UNICORN) hits the mechanical keyboard. Sparks fly.<br><br>
                        VEDRAN<br>
                        The chapters! Where are they?! The VHS Multiverse will collapse into a singularity of cheap 80s static!<br><br>
                        AETHER VEYLAN steps out of the shadows, spinning a glowing floppy disk on one finger.<br><br>
                        AETHER<br>
                        We didn't write them yet. We were too busy hot-gluing a lightbulb to a duck helmet.<br><br>
                        NARRATOR (V.O.)<br>
                        In a world where text files are missing, one duo must upload their stories before the 16-bit apocalypse hits. For risks and side effects, bargain with feta cheese.<br><br>
                        [SCENE END]
                    </div>`;
                }

                function buildNavigation() {
                    if (disposed) return;
                    navDiv.innerHTML = '';
                    for (let i = 1; i <= totalChapters; i++) {
                        const numStr = String(i).padStart(6, '0');
                        const btn = document.createElement('button');
                        btn.type = 'button';
                        btn.textContent = numStr;
                        btn.setAttribute('aria-label', `Kapitel ${i}`);
                        btn.addEventListener('click', () => loadChapter(i, true));
                        navDiv.appendChild(btn);
                    }
                }


                async function loadChapter(num, userInitiated = false) {
                    if (disposed) return;
                    if (loadedChapter === num) return;

                    currentChapter = num;
                    loadedChapter = 0;
                    const loadToken = ++chapterLoadToken;
                    navDiv.querySelectorAll('button').forEach((btn, index) => {
                        const isCurrent = index + 1 === num;
                        btn.classList.toggle('current', isCurrent);
                        if (isCurrent) btn.setAttribute('aria-current', 'page');
                        else btn.removeAttribute('aria-current');
                    });

                    contentDiv.innerHTML = '<span class="loading-text">LOADING FILE...</span>';
                    const numStr = String(num).padStart(6, '0');
                    cancelChapterFetch();
                    const controller = new AbortController();
                    fetchController = controller;

                    try {
                        const response = await fetch(`chapters/${numStr}.txt`, {
                            signal: controller.signal,
                            cache: 'no-store'
                        });
                        if (!response.ok) throw new Error(`HTTP ${response.status}`);
                        const rawText = await response.text();
                        if (disposed || loadToken !== chapterLoadToken) return;
                        parseChapterText(rawText);
                        loadedChapter = num;
                        
                        
                        
                        if (fetchController === controller) fetchController = null;
                        renderText();

                        
                        
                        const container = contentDiv.closest('.text-reader-container')
                            || document.querySelector('#mainContent .text-reader-container');
                        
                        

                        if (userInitiated && container) {
                            const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                            container.scrollIntoView({ behavior: reduceMotion ? 'auto' : 'smooth', block: 'start' });
                        }
                    } catch (error) {
                        if (error.name === 'AbortError' || disposed || loadToken !== chapterLoadToken) return;
                        contentDiv.innerHTML = '<p>CONNECTION ERROR: CHAPTER COULD NOT BE LOADED.</p>';
                    } finally {
                        if (fetchController === controller) fetchController = null;
                    }
                }

                function parseChapterText(rawText) {
                    if (rawText.includes('[ GERMAN VERSION ]')) {
                        const parts = rawText.split('[ GERMAN VERSION ]');
                        chapterData.en = parts[0].replace('[ ENGLISH VERSION ]', '').trim();
                        chapterData.de = parts[1].trim();
                    } else {
                        chapterData.en = rawText;
                        chapterData.de = 'ERROR 404: GERMAN TRANSLATION NOT FOUND IN ARCHIVE.';
                    }
                }

                function renderText() {
                    if (disposed) return;
                    
                    
                    if (fetchController !== null) return;
                    const mainText = currentLang === 'en' ? chapterData.en : chapterData.de;
                    const continuedText = currentLang === 'en' ? '\n\n\n... TO BE CONTINUED ...' : '\n\n\n... FORTSETZUNG FOLGT ...';
                    contentDiv.textContent = mainText + continuedText;
                    
                    
                    contentDiv.lang = currentLang;
                    const status = document.getElementById('storytime-status');
                    if (status) status.textContent = loadedChapter
                        ? `Chapter ${loadedChapter}` : '';
                }

                startupTimer = window.setTimeout(() => {
                    startupTimer = null;
                    scanChapters();
                }, 100);
            })();
        </script>
    </main>
